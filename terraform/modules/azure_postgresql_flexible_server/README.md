# Azure PostgreSQL Flexible Server Terraform Module

This Terraform module provisions an Azure Database for PostgreSQL Flexible Server. It wraps around the [Azure Verified Module for PostgreSQL Flexible Server](https://registry.terraform.io/modules/Azure/avm-res-dbforpostgresql-flexibleserver/azurerm/0.1.2) and includes additional customizations and validations.

## Introduction

This module simplifies the creation and management of Azure Database for PostgreSQL Flexible Servers with optional high availability, backup, and other settings. It integrates the core Azure Verified Module with additional validations for compliance with Azure's best practices.

---

## Usage Examples

### Basic Usage

```hcl
module "azure_postgres_flexible_server" {
  source              = "./IaC/terraform/modules/azure_postgres_flexible_server"
  location            = "eastus"
  name                = "my-postgres-server"
  resource_group_name = "my-resource-group"
  common_tags         = {
    environment = "production"
    project     = "my-project"
  }
}
```

### Complex Parameters Examples

#### High Availability Example

```hcl
module "azure_postgres_flexible_server" {
  source              = "./IaC/terraform/modules/azure_postgres_flexible_server"
  location            = "westus"
  name                = "my-postgres-server"
  resource_group_name = "my-resource-group"
  high_availability   = {
    mode                      = "ZoneRedundant"
    standby_availability_zone = "2"
  }
  common_tags = {
    environment = "production"
    project     = "my-project"
  }
}
```

#### Maintenance Window Example

```hcl
module "azure_postgres_flexible_server" {
  source              = "./IaC/terraform/modules/azure_postgres_flexible_server"
  location            = "eastus"
  name                = "my-postgres-server"
  resource_group_name = "my-resource-group"
  maintenance_window  = {
    day_of_week  = 2  # Tuesday
    start_hour   = 3  # 3 AM
    start_minute = 15 # 3:15 AM
  }
  common_tags = {
    environment = "staging"
    project     = "my-project"
  }
}
```

#### Private Endpoints Example

```hcl
module "azure_postgres_flexible_server" {
  source              = "./IaC/terraform/modules/azure_postgres_flexible_server"
  location            = "eastus"
  name                = "my-postgres-server"
  resource_group_name = "my-resource-group"
  private_endpoints   = {
    pe1 = {
      name = "private-endpoint-1"
      subnet_resource_id = "/subscriptions/<subscription_id>/resourceGroups/my-group/providers/Microsoft.Network/virtualNetworks/my-vnet/subnets/my-subnet"
      tags = {
        key1 = "value1"
      }
    }
  }
  common_tags = {
    environment = "production"
    project     = "my-project"
  }
}
```

#### DNS Zone with Private Endpoint Example

```hcl
module "azure_postgres_flexible_server" {
  source              = "./IaC/terraform/modules/azure_postgres_flexible_server"

  # Required inputs
  location            = "eastus"
  name                = "my-postgres-server"
  resource_group_name = "my-resource-group"
  common_tags         = {
    environment = "production"
    project     = "my-project"
  }

  # Optional inputs for private access configuration
  delegated_subnet_id = "/subscriptions/<subscription_id>/resourceGroups/my-group/providers/Microsoft.Network/virtualNetworks/my-vnet/subnets/my-delegated-subnet"
  public_network_access_enabled = false

  # Private DNS zone configuration
  private_dns_zone_id = "/subscriptions/<subscription_id>/resourceGroups/my-group/providers/Microsoft.Network/privateDnsZones/privatelink.postgres.database.azure.com"

  # Private endpoint configuration for VNet integration
  private_endpoints = {
    postgres_endpoint = {
      name                = "postgres-private-endpoint"
      subnet_resource_id  = "/subscriptions/<subscription_id>/resourceGroups/my-group/providers/Microsoft.Network/virtualNetworks/my-vnet/subnets/my-private-endpoint-subnet"
      private_dns_zone_group_name = "default"  # Will associate private DNS with this endpoint
      private_dns_zone_resource_ids = [
        "/subscriptions/<subscription_id>/resourceGroups/my-group/providers/Microsoft.Network/privateDnsZones/privatelink.postgres.database.azure.com"
      ]
      tags = {
        purpose = "postgres-private-access"
      }
    }
  }

  # Optional settings for backup and high availability
  backup_retention_days         = 30
  geo_redundant_backup_enabled  = true
  high_availability = {
    mode = "ZoneRedundant"
  }
}
```
